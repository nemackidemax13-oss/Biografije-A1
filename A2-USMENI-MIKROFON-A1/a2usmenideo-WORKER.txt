  async fetch(request, env) {
    const path = new URL(request.url).pathname.replace(/\/+$/, '') || '/';
    const cors = corsHeaders(request, env);
    const origin = request.headers.get('Origin');

    // An Origin header identifies browser requests, not a secure identity.
    if (origin && !cors) return json({ ok: false, error: 'Origin nije dozvoljen.' }, 403);

    if (request.method === 'OPTIONS') {
      return new Response(null, {
        status: 204,
        headers: { ...cors, 'Access-Control-Max-Age': '86400' }
      });
    }

    if (request.method === 'GET' && (path === '/' || path === '/health' || path === '/api/a2-usmeni')) {
      return json({
        ok: true,
        configured: typeof env.OPENAI_API_KEY === 'string' && env.OPENAI_API_KEY.trim().length > 10,
        worker: 'a2usmenideo',
        version: VERSION,
        vocabulary: { names: KNOWN_NAMES.length, cities: KNOWN_PLACES.length, priorityNames: PRIORITY_NAMES },
        transcription: string(env.TRANSCRIBE_MODEL, 80) || 'gpt-transcribe',
        endpoints: ['/api/a2-usmeni', '/api/a1-transcribe', '/api/a2-transcribe'],
        message: 'De-Max A2 Worker je aktivan. Proverite configured=true za AI funkcije.'
      }, 200, cors);
    }

    if (request.method !== 'POST') {
      return json({ ok: false, error: 'Nepoznata putanja ili metoda.' }, 404, cors);
    }

    if (!env.OPENAI_API_KEY || !String(env.OPENAI_API_KEY).trim()) {
      return json({ ok: false, error: 'Nedostaje OPENAI_API_KEY Secret na Worker-u.' }, 503, cors);
    }

    try {
      if (path === '/api/a1-transcribe' || path === '/api/a2-transcribe') {
        return await transcribe(request, env, cors);
      }
      if (path === '/api/a2-usmeni') {
        return await processAI(request, env, cors);
      }
      return json({ ok: false, error: 'Nepoznata API putanja.' }, 404, cors);
    } catch (error) {
      // Do not expose OpenAI headers, API keys, or internal request details.
      const code = Number(error?.status) || 500;
      const status = [400, 413, 415, 422, 429, 502, 503, 504].includes(code) ? code : 502;
      return json({ ok: false, error: publicError(error, status) }, status, cors);
    }
  }

function corsHeaders(request, env) {
  const origin = request.headers.get('Origin');
  const allowed = String(env.ALLOWED_ORIGINS || DEFAULT_ORIGIN)
    .split(',').map(x => x.trim()).filter(Boolean);
  if (origin && !allowed.includes(origin)) return null;
  return {
    ...(origin ? { 'Access-Control-Allow-Origin': origin } : {}),
    'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type',
    'Vary': 'Origin'
  };
}

function json(obj, status = 200, cors = {}) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: {
      'Content-Type': 'application/json; charset=utf-8',
      'Cache-Control': 'no-store',
      'X-Content-Type-Options': 'nosniff',
      ...(cors || {})
    }
  });
}

function fail(message, status = 400) {
  // Object.assign types the error with the HTTP status property.
  // Avoids Cloudflare editor TS2339 (Property 'status' does not exist on type 'Error').
  throw Object.assign(new Error(message), { status });
}

function publicError(error, status) {
  const text = String(error?.message || '').slice(0,240)
    .replace(/sk-(?:proj-)?[A-Za-z0-9_-]{10,}/g, '[REDACTED]');
  if (status < 500) return text || 'Neispravan zahtev.';
  if (status === 503) return text || 'API nije konfigurisan ili privremeno nije dostupan.';
  if (status === 504) return 'AI servis nije odgovorio na vreme.';
  return text || 'AI servis trenutno nije dostupan.';
}

function string(value, limit = 3500) {
  return typeof value === 'string' ? value.trim().slice(0, limit) : '';
}

function arrayOfStrings(value, max = 12) {
  return Array.isArray(value)
    ? value.slice(0, max).filter(x => typeof x === 'string').map(x => string(x, 100))
    : [];
}

function normal(value) {
  return String(value || '').normalize('NFKD').replace(/[\u0300-\u036f]/g, '')
    .toLocaleLowerCase('de').replace(/[^a-z0-9äöüß ]/g, '').replace(/\s+/g, ' ').trim();
}

async function processAI(request, env, cors) {
  const contentType = request.headers.get('Content-Type') || '';
  if (!contentType.includes('application/json')) fail('Očekujem application/json.', 415);
  if (Number(request.headers.get('Content-Length') || 0) > MAX_JSON_BYTES) {
    fail('JSON zahtev je prevelik.', 413);
  }
  const text = await request.text();
  if (new TextEncoder().encode(text).length > MAX_JSON_BYTES) fail('JSON zahtev je prevelik.', 413);
  let body;
  try { body = JSON.parse(text); } catch { fail('JSON nije ispravan.', 400); }
  if (!body || typeof body !== 'object' || Array.isArray(body)) fail('Očekujem JSON objekat.', 400);

  const type = string(body.type, 40);
  let result;
  if (type === 'evaluate') result = await evaluate(body, env);
  else if (type === 'translate') result = await translate(body, env);
  else if (type === 'generate_questions') result = await questions(body, env);
  else if (type === 'match_topics') result = await matchTopics(body, env);
  else if (type === 'correct_transcript') result = await correctTranscript(body, env);
  else if (type === 'finalize_answer') result = await finalizeAnswer(body, env);
  else if (type === 'review_grammar_batch') result = await reviewGrammarBatch(body, env);
  else fail('Nepoznata operacija: ' + type, 400);

  // Existing A2 app accepts either an object or an object inside {result:...}.
  return json({ ok: true, result, ...result }, 200, cors);
}

async function callJSON(env, system, data, maxTokens = 1300) {
  const model = string(env.TEXT_MODEL, 80) || 'gpt-5-mini';
  const payload = {
    model,
    messages: [
      { role: 'system', content: system + '\nUVEK vrati samo jedan ispravan JSON objekat.' },
      { role: 'user', content: JSON.stringify(data) }
    ],
    response_format: { type: 'json_object' },
    max_completion_tokens: maxTokens
  };
  if (/^gpt-5/i.test(model)) payload.reasoning_effort = 'low';

  const res = await openAI(env, '/chat/completions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });
  const answer = res.choices?.[0]?.message?.content;
  if (typeof answer !== 'string' || !answer.trim()) fail('Prazan odgovor AI servisa.', 502);
  try { return JSON.parse(answer); } catch { fail('AI odgovor nije ispravan JSON.', 502); }
}

async function openAI(env, path, options) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS);
  try {
    const res = await fetch(API_BASE + path, {
      ...options,
      signal: controller.signal,
      headers: { ...options.headers, Authorization: 'Bearer ' + env.OPENAI_API_KEY }
    });
    const value = await res.json().catch(() => ({}));
    if (!res.ok) {
      if (res.status === 429) fail('AI kvota je privremeno potrošena.', 429);
      if (res.status === 401 || res.status === 403) fail('API ključ nije prihvaćen. Proverite OPENAI_API_KEY.', 503);
      if (res.status === 400 || res.status === 404 || res.status === 422) {
        // Detalji kao što su nepoznat model ili nepodržan parametar
        // su potrebni za dijagnostiku. Nikada ne vraćaj API ključ.
        const raw = string(value?.error?.message, 220);
        const safe = raw.replace(/sk-(?:proj-)?[A-Za-z0-9_-]{10,}/g, '[redacted]');
        fail('AI zahtev je odbijen (HTTP ' + res.status + '): ' + (safe || 'Proverite model i parametre.'), 422);
      }
      if (res.status >= 500) fail('OpenAI trenutno nije dostupan (HTTP ' + res.status + ').', 503);
      fail('AI servis je vratio HTTP ' + res.status + '.', 502);
    }
    return value;
  } catch (error) {
    if (error?.name === 'AbortError') fail('AI servis nije odgovorio na vreme.', 504);
    throw error;
  } finally {
    clearTimeout(timer);
  }
}

/** One AI request to correct clear ASR mistakes AND translate the full final turn.
 *  This avoids sequential correction then translation after five seconds of silence.
 */
async function finalizeAnswer(body, env) {
  const original = string(body.transcript || body.text || body.answer, 8000);
  if (!original) return { original: '', corrected: '', changed: false, uncertain: false, translationSr: '', suggestions: [] };
  const topic = string(body.topic, 150);
  const question = string(body.question, 500);
  const result = await callJSON(env,
    'Ti si precizan prevodilac i OPREZAN korektor transkripcije nemačkog govora A2. ' +
    'Korisnik je upravo završio svoj govor. U JEDNOM odgovoru vrati: ' +
    '1) corrected: minimalno korigovan nemački transkript SAMO gde je greška ASR-a sasvim očigledna, ' +
    '2) uncertain: true ako bi ispravljanje zahtevalo nagađanje, ' +
    '3) translationSr: veran i POTPUN prevod izgovorenog nemačkog na srpski latinicom, ' +
    '4) suggestions: najviše tri moguće ispravke kada postoji neizvesnost. ' +
    'NEMOJ dopunjavati rečenice, menjati negacije, gradove, imena, brojeve, ' +
    'datum, godine, pol niti menjati ono što učenik jeste rekao. ' +
    'Očigledne pravopisne greške imena mesta (Tutin, Novi Pazar, Sjenica, Kraljevo, Belgrad, Beograd, Podgorica) ' +
    'i izgovorenih ličnih imena (Naida, Ema, Ajlina, Merima) ' +
    'ispravi samo kada postoji dovoljno konteksta. ' +
    'Primer "Ich komme auszuziehen" možda znači "Ich komme aus Tutin", ' +
    'ali SAMO tekst nije dokaz; ako je neizvesno, vrati uncertain=true ' +
    'i corrected identičan izvornom transkriptu. ' +
    'TranslationSr treba da odgovara corrected samo ako je izmena sigurna, ' +
    'inače originalu. Ako je neka reč nerazumljiva, zadrži je u navodnicima ' +
    'umesto izmišljanja. Nijedan uzorni ispitni odgovor ne sme da se ubaci. ' +
    'Vrati JSON {"corrected":"...","uncertain":false,"translationSr":"...","suggestions":[]}.',
    { original, topic, question, candidatePlaces: KNOWN_PLACES,
      candidateNames: KNOWN_NAMES, priorityNames: PRIORITY_NAMES,
      note: 'Rečnik su samo mogući zapisi, nisu dokazi da su imena izgovorena.' },
    1900);
  const uncertain = result.uncertain === true;
  const proposed = string(result.corrected, 8000) || original;
  const safe = !uncertain && Math.abs(proposed.length - original.length) <= Math.max(45, original.length * 0.35);
  const corrected = safe ? proposed : original;
  const suggestions = arrayOfStrings(result.suggestions, 3);
  const translationSr = string(result.translationSr, 8000);
  if (!translationSr) fail('AI nije vratio prevod završnog odgovora.', 502);
  // Never keep Serbian translation of an unsafe German rewrite.
  if (!safe && proposed !== original && !uncertain) {
    const strict = await translate({ text: original }, env);
    return { original, corrected: original, changed: false, uncertain: true, suggestions, translationSr: strict.translationSr };
  }
  return { original, corrected, changed: corrected !== original, uncertain, suggestions, translationSr };
}

// Final exam report: review exactly the student's transcript, 1-5 entries per call.
// Keep the original transcript unchanged; return only confident grammatical corrections.
async function reviewGrammarBatch(body, env) {
  const input = Array.isArray(body.items) ? body.items.slice(0, 5) : [];
  const items = input.map(item => ({
    id: string(item?.id, 90),
    text: string(item?.text, 1800),
    question: string(item?.question, 350),
    topic: string(item?.topic, 120)
  })).filter(item => item.id && item.text);
  if (!items.length) return { reviews: [] };
  if (items.some(item => !/^[a-zA-Z0-9_-]{1,90}$/.test(item.id))) fail('Neispravan ID odgovora.', 400);
  const result = await callJSON(env,
    'Ti proveravaš gramatičke greške u ZAPISANOM nemačkom govoru učenika A2. ' +
    'Nemaš pristup audio-zapisu: ne ocenjuj izgovor i ne tvrdi da si čuo tačne reči. ' +
    'Za SVAKU stavku vrati isti id, hasError, corrected, explanation, uncertain. ' +
    'Ovo NIJE pisanje model-odgovora. Originalni tekst mora ostati identičan u korisničkom izveštaju. ' +
    'Ako nema jasne gramatičke greške, hasError=false, corrected bude isti kao text, explanation prazno. ' +
    'Ako ima jasne gramatičke greške, ispravi SAMO tu grešku u celoj učenikovoj rečenici, ' +
    'sačuvaj identične činjenice, redosled, imena, gradove, brojeve, negacije i nameru govornika. ' +
    'Ne dopunjavaj prekinute rečenice, nemoj odgovarati na pitanje umesto učenika, ' +
    'ne dodaj drugu rečenicu, ne zamenjuj ljude niti izmišljaj događaje. ' +
    'Ako tekst može biti greška prepoznavanja govora, ili korekcija traži pretpostavke, ' +
    'uncertain=true, hasError=false, corrected isti kao text. ' +
    'explanation je kratko objašnjenje STVARNE gramatičke greške na srpskom latinicom; ' +
    'bez lažnih komentara o izgovoru ili pravopisu vlastitih imena. ' +
    'Vrati JSON {"reviews":[{"id":"...","hasError":false,"corrected":"...","explanation":"","uncertain":false}]}. ' +
    'Ne preskači nijedan id, ne dodaj izmišljene stavke.',
    { items },
    2400);
  const rows = Array.isArray(result.reviews) ? result.reviews : [];
  const used = new Set();
  const reviews = [];
  for (const item of items) {
    const row = rows.find(x => x?.id === item.id && !used.has(item.id));
    if (!row) continue;
    used.add(item.id);
    const proposed = string(row.corrected, 2200);
    const changed = proposed && normal(proposed) !== normal(item.text);
    // Do not accept wholesale rewrites of student speech as corrections.
    const reasonable = Math.abs(proposed.length - item.text.length) <=
      Math.max(28, Math.ceil(item.text.length * 0.42));
    const negations = x => (String(x).match(/\b(?:nicht|kein|keine|keinen|keinem|keiner|nie|niemals|ohne)\b/gi) || [])
      .map(v => v.toLowerCase()).sort().join('|');
    const numbers = x => (String(x).match(/\d+(?:[.,]\d+)*/g) || []).sort().join('|');
    const safe = row.uncertain !== true && row.hasError === true &&
      changed && reasonable && negations(item.text) === negations(proposed) &&
      numbers(item.text) === numbers(proposed);
    reviews.push({
      id: item.id, checked: true,
      hasError: !!safe,
      corrected: safe ? proposed : item.text,
      explanation: safe ? string(row.explanation, 350) : '',
      uncertain: row.uncertain === true || (row.hasError === true && !safe)
    });
  }
  return { reviews };
}

    return {
      score: 0,
      feedback: 'Učenik nije odgovorio.',
      corrected: '',
      suggestedAnswer: '',
      translationSr: ''
    };
  }
  const result = await callJSON(env,
    'Ti si pravedan ispitivač nemačkog A2 za ÖSD usmeni ispit. ' +
    'Oceni SAMO stvarni odgovor učenika prema pitanju, smislu, gramatici i razumljivosti; ' +
    'ne kažnjavaj dijalekt ni previše strogo sitne greške. ' +
    'Nemaš zvuk: ne ocenjuj izgovor. Ne pretpostavljaj pol učenika. ' +
    'Score je broj 0, 0.5 ili 1. Kada je section=topic ili intro, score služi samo kao orijentir. ' +
    'feedback piši kratko na srpskom latinicom. corrected je gramatički ispravljen učenikov ' +
    'odgovor bez dodavanja novih činjenica. suggestedAnswer je kratak A2 primer relevantan pitanju. ' +
    'translationSr je VERAN PREVOD ISKLJUČIVO izgovorenog/naučenog učenikovog answer, ' +
    'nikako ispravljenog ili predloženog odgovora. ' +
    'Vrati JSON {"score":0,"feedback":"","corrected":"","suggestedAnswer":"","translationSr":""}.',
    { section, topic, question, answer }, 1800);
  const numeric = Number(result.score);
  const score = Number.isFinite(numeric) ? Math.max(0, Math.min(1, Math.round(numeric * 2) / 2)) : 0;
  return {
    score,
    feedback: string(result.feedback, 750) || 'Odgovor je pregledan.',
    corrected: string(result.corrected, 1200),
    suggestedAnswer: string(result.suggestedAnswer, 1200),
    translationSr: string(result.translationSr, 8000)
  };
}

async function questions(body, env) {
  const topic = string(body.topic, 120);
  const avoid = arrayOfStrings(body.avoid, 20);
  const count = Math.min(5, Math.max(1, Number(body.count) || 2));
  if (!topic) fail('Nedostaje topic.', 400);
  const result = await callJSON(env,
    'Sastavi kratka i prirodna pitanja za usmeni ÖSD A2. ' +
    'Neka odgovori budu mogući u 1–2 jednostavne rečenice. ' +
    'Nema pitanja o supružniku ako je na spisku zabranjenih reči. ' +
    'Ne pretpostavljaj pol, bračni status, roditeljstvo niti činjenice o učeniku. ' +
    'Nemoj ponavljati isto pitanje. ' +
    'Vrati JSON {"questions":[{"q":"pitanje na nemačkom",' +
    '"qt":"prevod pitanja na srpski latinicom",' +
    '"a":"kratak primer odgovora na nemačkom",' +
    '"at":"prevod primera na srpski latinicom"}]}.',
    { topic, count, avoid }, 1800);
  const all = Array.isArray(result.questions) ? result.questions : [];
  const unique = new Set();
  const normalizedAvoid = avoid.map(normal);
  const questions = all.map(item => ({
    q: string(item?.q, 350), qt: string(item?.qt, 450),
    a: string(item?.a, 450), at: string(item?.at, 450)
  })).filter(item => {
    const k = normal(item.q);
    if (!k || unique.has(k)) return false;
    if (normalizedAvoid.some(word => word && (' ' + k + ' ').includes(' ' + word + ' '))) return false;
    unique.add(k); return true;
  }).slice(0, count);
  return { questions };
}

async function matchTopics(body, env) {
  const transcript = string(body.transcript || body.text, 3000);
  const candidates = arrayOfStrings(body.candidates, 12);
  const maxSelections = Math.max(1, Math.min(5, Number(body.maxSelections) || 5));
  if (!transcript || !candidates.length) return { topics: [] };
  const result = await callJSON(env,
    'Razumeš izgovorena imena tema u nemačkom A2 usmenom ispitu. ' +
    'Vrati samo izgovorene teme iz ponuđenog candidates, u redosledu kojim su pomenute. ' +
    'Dopuštene su očigledne fonetske varijacije. ' +
    'Ne dodaj nijednu temu koja nije rečena, i ne dopunjavaj listu do pet. ' +
    'Vrati JSON {"topics":["tema 1","tema 2"]}.',
    { transcript, candidates, maxSelections }, 900);
  const seen = new Set();
  const topics = (Array.isArray(result.topics) ? result.topics : []).map(item => {
    const value = typeof item === 'string' ? item : item?.name;
    return candidates.find(c => normal(c) === normal(value));
  }).filter(item => {
    if (!item || seen.has(item)) return false;
    seen.add(item); return true;
  }).slice(0, maxSelections);
  return { topics };
}

async function correctTranscript(body, env) {
  const original = string(body.transcript || body.text || body.answer, 8000);
  if (!original) return { original: '', corrected: '', changed: false, uncertain: false, suggestions: [] };
  const topic = string(body.topic, 140);
  const question = string(body.question, 500);
  const candidates = arrayOfStrings(body.places, 20);
  const result = await callJSON(env,
    'Ti si izuzetno oprezan korektor NEMAČKOG automatskog prepoznavanja govora, A2 nivo. ' +
    'Ispravi SAMO veoma očigledne greške ASR-a. Posebno obrati pažnju na geografska ' +
    'imena Tutin, Novi Pazar, Sjenica, Kraljevo, Belgrad (Beograd) i Podgorica. ' +
    'Lična imena Naida, Ema, Ajlina i Merima pišu se ovako kada su zaista izgovorena. ' +
    'Primer rečenice: "Ich komme aus Tutin" i "Ich wohne in Tutin". ' +
    'Primer: "Ich komme auszuziehen" može značiti "Ich komme aus Tutin", ' +
    'ALI to ne možeš potvrditi bez zvuka; u tom slučaju stavi uncertain=true i ' +
    'ponudi suggestions umesto da nametneš zamenu kao sigurnu činjenicu. ' +
    'Ne menjaj ispravni glagol "auszuziehen" u rečenici o selidbi. ' +
    'Sačuvaj stvarni jezik učenika, čak i ako je gramatički pogrešan. ' +
    'Ne dodaj ništa i ne prepravljaj ime osobe ili grada bez dokaza. ' +
    'Vrati JSON {"corrected":"tekst", "uncertain":false, "suggestions":["alternativa"]}.',
    { original, topic, question, candidatePlaces: candidates.length ? [...new Set([...candidates,...KNOWN_PLACES])] : KNOWN_PLACES,
      candidateNames: KNOWN_NAMES, priorityNames: PRIORITY_NAMES,
      note: 'Ispraviti samo ako fonetski ili kontekst jasno potvrđuje ime.' }, 1700);
  const corrected = string(result.corrected, 8000) || original;
  const uncertain = result.uncertain === true;
  const suggestions = arrayOfStrings(result.suggestions, 5);
  // Uncertain meaning must never silently replace the learner's original words.
  const finalText = uncertain ? original : corrected;
  return { original, corrected: finalText, changed: finalText !== original, uncertain, suggestions };
}

async function transcribe(request, env, cors) {
  const contentType = request.headers.get('Content-Type') || '';
  if (!contentType.toLowerCase().includes('multipart/form-data')) {
    fail('Pošaljite FormData sa poljem file.', 415);
  }
  if (Number(request.headers.get('Content-Length') || 0) > MAX_AUDIO_BYTES + 200_000) {
    fail('Audio snimak je prevelik.', 413);
  }
  let body;
  try { body = await request.formData(); } catch { fail('Neispravan audio FormData.', 400); }
  const file = body.get('file');
  // Ne koristi instanceof File: u nekim Cloudflare Worker okruženjima
  // globalni File nije dostupan ili upload koristi drugu implementaciju.
  // request.formData() vraća File-like podatke sa .size i .stream().
  if (!file || typeof file === 'string' || typeof file.size !== 'number' ||
      typeof file.stream !== 'function' || file.size < 500) {
    fail('Nedostaje audio fajl ili je snimak prazan.', 400);
  }
  if (file.size > MAX_AUDIO_BYTES) fail('Audio snimak prelazi dozvoljenu veličinu.', 413);
  const extension = String(file.name || '').toLowerCase().split('.').pop();
  const content = String(file.type || '').toLowerCase();
  if (!['webm', 'mp4', 'm4a', 'mp3', 'wav', 'mpeg', 'mpga'].includes(extension)
      && !/(audio\/(webm|mp4|mpeg|mp3|wav|x-wav|m4a)|video\/mp4)/.test(content)) {
    fail('Format zvuka nije podržan. Koristite webm, mp4, m4a, mp3 ili wav.', 415);
  }

  const form = new FormData();
  const ext = ['webm', 'mp4', 'm4a', 'mp3', 'wav', 'mpeg', 'mpga'].includes(extension)
    ? extension : (content.includes('mp4') ? 'mp4' : 'webm');
  form.append('file', file, 'nemacki-odgovor.' + ext);
  form.append('model', string(env.TRANSCRIBE_MODEL, 80) || 'gpt-transcribe');
  const context = string(body.get('context'), 240);
  const model = string(env.TRANSCRIBE_MODEL, 80) || 'gpt-transcribe';
  // U svakom snimku daj prednost traženim imenima i gradovima.
  // U porodici/predstavljanju proširi izbor na ravnomerno uzorkovana
  // bošnjačka i srpska imena. Ne šalji stotine termina u jednom snimku.
  const nameTopic = /famil|kind|name|vorstell|gebor|person|eltern|tochter|sohn/i.test(context);
  const names = nameTopic
    ? [...PRIORITY_NAMES, ...BOSNIAK_NAMES.slice(0, 18), ...SERBIAN_NAMES.slice(0, 18)]
    : [...PRIORITY_NAMES, ...BOSNIAK_NAMES.slice(0, 8), ...SERBIAN_NAMES.slice(0, 8)];
  const keywords = [...new Set([...KNOWN_PLACES.slice(0, 16), ...names])]
    .filter(item => item && !/[<>\r\n]/.test(item)).slice(0, nameTopic ? 50 : 36);
  form.append('prompt',
    'ÖSD A2, mündliche Prüfung auf Deutsch. Schreibe die tatsächlich gesprochenen Wörter ' +
    'mit den korrekten Namen, ohne etwas hinzuzufügen. Mögliche Vornamen: ' +
    'Naida, Ema, Ajlina, Merima. Mögliche Ortsnamen: Tutin, Novi Pazar, Sjenica, ' +
    'Kraljevo, Belgrad, Beograd, Podgorica. ' +
    'Nur Namen verwenden, die im Audio tatsächlich gesprochen werden. ' +
    'Bei Stille oder unverständlichen Geräuschen leer transkribieren. ' +
    'Keine Beispielsätze übernehmen, nichts dazuerfinden, keine Sätze vervollständigen. ' +
    'Grammatikfehler des Lernenden nicht verbessern.' +
    (context ? ' Prüfungsthema: ' + context.replace(/[<>\r\n]/g, ' ') + '.' : ''));
  if (model === 'gpt-transcribe') {
    form.append('languages[]', 'de');
    for (const term of keywords) form.append('keywords[]', term);
  } else {
    form.append('language', 'de');
  }

  const upstream = await openAI(env, '/audio/transcriptions', {
    method: 'POST', body: form
    // Never set multipart Content-Type manually; fetch generates the boundary.
  });
  const raw = string(upstream.text, 12000);
  return json({ ok: true, text: raw, transcript: raw, language: 'de' }, 200, cors);
}
